     8→ *   - جدول‌های بدون کلید یونیک (Slide/Banner/Faq) با deleteMany + create
     9→ *   - فرزندان Product (تصاویر/مشخصات/تنوع‌ها) در مسیر update با deleteMany + create بازسازی می‌شوند

    26→/** نرمال‌سازی متن جست‌وجو: ي/ك عربی→فارسی، حذف نیم‌فاصله‌های تکراری، lower-case انگلیسی، فشرده‌سازی فاصله‌ها */
    27→function normalizeSearchText(s: string): string {
    28→  return s
    29→    .replace(/ي/g, "ی")
    30→    .replace(/ك/g, "ک")
    31→    .replace(/\u200C+/g, "\u200C")
    32→    .replace(/\s+/g, " ")
    33→    .toLowerCase()
    34→    .trim();
    35→}
    36→
    37→const DAY_MS = 86_400_000;
    38→const now = Date.now();
    39→
    40→/** createdAt محصولات: پخش خطی روی ۶۰ روز اخیر (ایندکس ۰ → ۶۰ روز پیش، ایندکس ۱۴ → امروز) */
    41→function productCreatedAt(index: number): Date {
    42→  return new Date(now - (60 - (index * 60) / 14) * DAY_MS);
    43→}
    44→
    45→function daysAgo(days: number): Date {
    46→  return new Date(now - days * DAY_MS);
    47→}

   781→
   782→  for (const [parentIndex, parent] of CATEGORIES.entries()) {

   784→    const parentRow = await prisma.category.upsert({
   785→      where: { slug: parent.slug },
   786→      create: {
   787→        title: parent.title,
   788→        slug: parent.slug,
   789→        iconKey: parent.iconKey,
   790→        sortOrder: parentIndex,
   791→        isActive: true,
   792→        seoTitle,
   793→        seoDescription: parent.seoDescription,
   794→      },
   795→      update: {
   796→        title: parent.title,
   797→        iconKey: parent.iconKey,
   798→        sortOrder: parentIndex,
   799→        isActive: true,
   800→        seoTitle,
   801→        seoDescription: parent.seoDescription,
   802→        parentId: null,
   803→      },
   804→    });
   805→    map.set(parent.slug, parentRow);
   806→
   807→    for (const [childIndex, child] of parent.children.entries()) {
   808→      const childRow = await prisma.category.upsert({
   809→        where: { slug: child.slug },
   810→        create: {
   811→          title: child.title,
   812→          slug: child.slug,
   813→          iconKey: null,
   814→          sortOrder: childIndex,
   815→          isActive: true,
   816→          parentId: parentRow.id,
   817→        },
   818→        update: {
   819→          title: child.title,
   820→          iconKey: null,
   821→          sortOrder: childIndex,
   822→          isActive: true,
   823→          parentId: parentRow.id,
   824→        },
   825→      });
   826→      map.set(child.slug, childRow);
   827→    }
   828→  }

   845→async function seedProducts(
   846→  categoryMap: Map<string, Category>,
   847→  brandMap: Map<string, Brand>,
   848→): Promise<void> {

   865→      where: { slug: p.slug },

   877→        isFeatured: p.isFeatured,

   879→        categoryId: category.id,
   880→        brandId: brand.id,
   881→        images: { create: images },

   883→        variants: p.variants ? { create: p.variants } : undefined,

   885→      update: {
