   835→   835→    const row = await prisma.brand.upsert({

   838→   838→      update: { title: brand.title, logoUrl: null, isActive: true, sortOrder: index },

   845→   845→async function seedProducts(
   846→   846→  categoryMap: Map<string, Category>,
   847→   847→  brandMap: Map<string, Brand>,
   848→   848→): Promise<void> {
   849→   849→  for (const [index, p] of PRODUCTS.entries()) {
   850→   850→    const category = categoryMap.get(p.categorySlug);
   851→   851→    const brand = brandMap.get(p.brandSlug);
   852→   852→    if (!category || !brand) {
   853→   853→      throw new Error(`دسته یا برند برای محصول «${p.slug}» پیدا نشد.`);
   854→   854→    }
   855→   855→
   856→   856→    const searchText = normalizeSearchText(`${p.title} ${brand.title}`);
   857→   857→    const createdAt = productCreatedAt(index);
   858→   858→    const images = [
   859→   859→      { url: `/placeholders/product-${index + 1}-a.svg`, alt: p.title, sortOrder: 0 },
   860→   860→      { url: `/placeholders/product-${index + 1}-b.svg`, alt: p.title, sortOrder: 1 },
   861→   861→    ];
   862→   862→    const specs = p.specs.map((s, i) => ({ key: s.key, value: s.value, sortOrder: i }));
   863→   863→
   864→   864→    await prisma.product.upsert({
   865→   865→      where: { slug: p.slug },
   866→   866→      create: {
   867→   867→        title: p.title,
   868→   868→        slug: p.slug,
   869→   869→        searchText,
   870→   870→        shortDescription: p.shortDescription,
   871→   871→        description: p.description,
   872→   872→        price: p.price,
   873→   873→        comparePrice: p.comparePrice,
   874→   874→        stock: p.stock,
   875→   875→        isActive: true,
   876→   876→        isNew: p.isNew,
   877→   877→        isFeatured: p.isFeatured,
   878→   878→        createdAt,
   879→   879→        categoryId: category.id,
   880→   880→        brandId: brand.id,
   881→   881→        images: { create: images },
   882→   882→        specs: { create: specs },
   883→   883→        variants: p.variants ? { create: p.variants } : undefined,
   884→   884→      },
   885→   885→      update: {
   886→   886→        title: p.title,
   887→   887→        searchText,
   888→   888→        shortDescription: p.shortDescription,
   889→   889→        description: p.description,
   890→   890→        price: p.price,
   891→   891→        comparePrice: p.comparePrice,
   892→   892→        stock: p.stock,
   893→   893→        isActive: true,
   894→   894→        isNew: p.isNew,
   895→   895→        isFeatured: p.isFeatured,
   896→   896→        createdAt,
   897→   897→        categoryId: category.id,
   898→   898→        brandId: brand.id,
   899→   899→        images: { deleteMany: {}, create: images },
   900→   900→        specs: { deleteMany: {}, create: specs },
   901→   901→        variants: p.variants ? { deleteMany: {}, create: p.variants } : { deleteMany: {} },
   902→   902→      },
   903→   903→    });
   904→   904→  }
   905→   905→}
