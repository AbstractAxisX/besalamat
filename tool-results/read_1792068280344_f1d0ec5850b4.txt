   584→/** ساخت ProductSummary از ردیف seed (id ها slug-محور: p-<slug>) */
   585→function toProductSummary(row: PreviewProductRow, index: number): ProductSummary {
   586→  return {
   587→    id: `p-${row.slug}`,
   588→    slug: row.slug,
   589→    title: row.title,
   590→    image: `/placeholders/product-${index + 1}-a.svg`,
   591→    price: row.price,
   592→    comparePrice: row.comparePrice,
   593→    inStock: row.totalStock > 0,
   594→    brandTitle: row.brandTitle,
   595→    isNew: row.isNew,
   596→    isFeatured: row.isFeatured,
   597→  } satisfies ProductSummary;
   598→}

   811→export const PRODUCT_TABS: { key: ProductTabKey; label: string }[] = [
   812→  { key: "bestseller", label: "پرفروش‌ها" },
   813→  { key: "new", label: "جدیدترین‌ها" },

   816→
   817→/** ترتیب دستی پرفروش‌ها — محصولات پرفروش‌تر اول (۸ مورد) */
   818→const BESTSELLER_SLUGS = [
   819→  "omron-m2-blood-pressure", // فشارسنج M2
   820→  "surgical-mask-50", // ماسک جراحی
   821→  "beurer-gl50-strips", // نوار قند
   822→  "beurer-ft90", // دماسنج FT90
   823→  "beurer-gs27", // ترازو GS27
   824→  "rossmax-tg300", // قندسنج TG300
   825→  "medisana-uhw", // بخور
   826→  "medisana-bs430", // آنالیزاتور بدن
   827→] as const;
   828→
   829→/**
   830→ * محصول‌های هر تب صفحهٔ خانه:
   831→ * - bestseller: ترتیب دستی منطقی ۸تایی
   832→ * - new: isNewها اول (تازه‌تر از seed اول‌تر)، سپس تازه‌ترین createdAtها تا ۸ مورد
   833→ *   (createdAt آینهٔ seed است: پخش خطی روی ۶۰ روز؛ ایندکس بالاتر = جدیدتر)
   834→ * - discount: فقط comparePrice != null، مرتب بر اساس درصد تخفیف (نزولی)
   835→ *   توجه: در seed فقط ۵ محصول comparePrice دارند؛ تب عین داده است.
   836→ */
   837→export function getPreviewProductsByTab(tab: ProductTabKey): ProductSummary[] {
   838→  switch (tab) {
   839→    case "bestseller": {
   840→      const bySlug = new Map(PREVIEW_PRODUCTS.map((product) => [product.slug, product]));
   841→      return BESTSELLER_SLUGS.map((slug) => bySlug.get(slug)).filter(
   842→        (product): product is ProductSummary => product != null,
   843→      );
   844→    }
   845→    case "new": {
   846→      return [...PREVIEW_PRODUCTS]
   847→        .map((product, index) => ({ product, index }))
   848→        .sort((a, b) => {
   849→          if (a.product.isNew !== b.product.isNew) return a.product.isNew ? -1 : 1;
   850→          return b.index - a.index;
   851→        })
   852→        .slice(0, 8)
   853→        .map((row) => row.product);
   854→    }
   855→    case "discount": {
   856→      return PREVIEW_PRODUCTS.map((product) => ({
   857→        product,
   858→        percent: discountPercent(product.price, product.comparePrice),
   859→      }))
   860→        .filter((row) => row.percent != null)
   861→        .sort((a, b) => (b.percent ?? 0) - (a.percent ?? 0))
   862→        .map((row) => row.product);
   863→    }
   864→  }
   865→}
