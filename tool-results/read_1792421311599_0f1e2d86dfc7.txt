   586→/** ساخت ProductSummary از ردیف seed (id ها slug-محور: p-<slug>) */
   587→function toProductSummary(row: PreviewProductRow, index: number): ProductSummary {
   588→  return {
   589→    id: `p-${row.slug}`,
   590→    slug: row.slug,
   591→    title: row.title,
   592→    image: `/placeholders/product-${index + 1}-a.svg`,
   593→    price: row.price,
   594→    comparePrice: row.comparePrice,
   595→    inStock: row.totalStock > 0,
   596→    brandTitle: row.brandTitle,
   597→    isNew: row.isNew,
   598→    isFeatured: row.isFeatured,
   599→    categorySlug: row.categorySlug,
   600→  } satisfies ProductSummary;
   601→}

   603→export const PREVIEW_PRODUCTS: ProductSummary[] = PREVIEW_PRODUCT_ROWS.map(toProductSummary);

   605→export const PREVIEW_PRODUCT_DETAILS: Record<string, ProductDetail> = Object.fromEntries(
   606→  PREVIEW_PRODUCT_ROWS.map((row, index) => {
   607→    const summary = toProductSummary(row, index);
   608→    return [
   609→      row.slug,
   610→      {
   611→        ...summary,
   612→        shortDescription: row.shortDescription,
   613→        description: row.description,
   614→        images: [
   615→          { url: `/placeholders/product-${index + 1}-a.svg`, alt: row.title },
   616→          { url: `/placeholders/product-${index + 1}-b.svg`, alt: row.title },

   619→          id: `pv-${row.slug}-${variantIndex + 1}`,

   626→        specs: row.specs,

   628→        categorySlug: row.categorySlug,
   629→        categoryParentTitle: row.categoryParentTitle,
   630→        totalStock: row.totalStock,
   631→      } satisfies ProductDetail,
   632→    ];
   633→  }),

   972→ * - discount: فقط comparePrice != null، مرتب بر اساس درصد تخفیف (نزولی)
   973→ *   توجه: در seed فقط ۵ محصول comparePrice دارند؛ تب عین داده است.
   974→ */
   975→export function getPreviewProductsByTab(tab: ProductTabKey): ProductSummary[] {
   976→  switch (tab) {
   977→    case "bestseller": {
   978→      const bySlug = new Map(PREVIEW_PRODUCTS.map((product) => [product.slug, product]));
   979→      return BESTSELLER_SLUGS.map((slug) => bySlug.get(slug)).filter(
   980→        (product): product is ProductSummary => product != null,
   981→      );
   982→    }
   983→    case "new": {
   984→      return [...PREVIEW_PRODUCTS]
   985→        .map((product, index) => ({ product, index }))
   986→        .sort((a, b) => {
   987→          if (a.product.isNew !== b.product.isNew) return a.product.isNew ? -1 : 1;
   988→          return b.index - a.index;
   989→        })
   990→        .slice(0, 8)
   991→        .map((row) => row.product);
   992→    }
   993→    case "discount": {
   994→      return PREVIEW_PRODUCTS.map((product) => ({
   995→        product,
   996→        percent: discountPercent(product.price, product.comparePrice),
   997→      }))
   998→        .filter((row) => row.percent != null)
   999→        .sort((a, b) => (b.percent ?? 0) - (a.percent ?? 0))
  1000→        .map((row) => row.product);
  1001→    }
  1002→  }
  1003→}
