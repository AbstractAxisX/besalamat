   105→/** نرمال‌سازی متن جستجو: ارقام لاتین + ی/ک یکسان‌شده */
   106→function normalizeText(value: string): string {
   107→  return toEnDigits(value)
   108→    .replace(/ي/g, "ی")
   109→    .replace(/ك/g, "ک")
   110→    .trim()
   111→    .toLowerCase();
   112→}
   113→
   114→/** مجموعهٔ اسلاگ‌های یک دستهٔ انتخابی — والد، فرزندانش را هم شامل می‌شود */
   115→function collectCategorySlugs(
   116→  categories: CategoryNode[],
   117→  slug: string,
   118→): Set<string> | null {
   119→  for (const parent of categories) {
   120→    if (parent.slug === slug) {
   121→      return new Set([parent.slug, ...parent.children.map((child) => child.slug)]);
   122→    }
   123→    if (parent.children.some((child) => child.slug === slug)) {
   124→      return new Set([slug]);
   125→    }
   126→  }
   127→  return null;
   128→}
   129→
   130→/** عنوان دستهٔ فعال (والد یا فرزند) */
   131→function findCategoryTitle(
   132→  categories: CategoryNode[],
   133→  slug: string,
   134→): string | null {
   135→  for (const parent of categories) {
   136→    if (parent.slug === slug) return parent.title;
   137→    const child = parent.children.find((item) => item.slug === slug);
   138→    if (child) return child.title;
   139→  }
   140→  return null;
   141→}
   142→
   143→/** پنجرهٔ صفحات: [۱، …، ۴، ۵، ۶، …، ۲۰] */
   144→function pageWindow(current: number, total: number): (number | "ellipsis")[] {
   145→  if (total <= 7) {
   146→    return Array.from({ length: total }, (_, index) => index + 1);
   147→  }
   148→  const items: (number | "ellipsis")[] = [1];
   149→  const start = Math.max(2, current - 1);
   150→  const end = Math.min(total - 1, current + 1);
   151→  if (start > 2) items.push("ellipsis");
   152→  for (let page = start; page <= end; page += 1) items.push(page);
   153→  if (end < total - 1) items.push("ellipsis");
   154→  items.push(total);
   155→  return items;
   156→}

   162→  /** slug محصول → دستهٔ فرزند (برای فیلتر دسته‌بندی) */
   163→  categoryByProduct: Record<string, string | null>;
   164→  initialFilters: CatalogInitialFilters;
   165→  /** صفحهٔ جستجو عنوان خودش را دارد و سرِ کاتالوگ را نمی‌خواهد */
   166→  showHeading?: boolean;

   174→
   175→export function CatalogView({
   176→  products,
   177→  categories,
   178→  brands,
   179→  categoryByProduct,
   180→  initialFilters,
   181→}: CatalogViewProps) {
   182→  const router = useRouter();
   183→  const pathname = usePathname();
   184→
   185→  const [filters, setFilters] = useState<CatalogFilterState>(() => ({
   186→    category: initialFilters.category ?? null,
   187→    brands:
   188→      initialFilters.brand == null
   189→        ? []
   190→        : Array.isArray(initialFilters.brand)
   191→          ? initialFilters.brand
   192→          : [initialFilters.brand],
   193→    min: initialFilters.min ?? null,
   194→    max: initialFilters.max ?? null,
   195→    inStock: initialFilters.inStock ?? false,
   196→    discount: initialFilters.discount ?? false,
   197→    q: initialFilters.q ?? null,
   198→  }));
   199→  const [sort, setSort] = useState<CatalogSortKey>(
   200→    initialFilters.sort ?? "newest",
   201→  );
   202→  const [view, setView] = useState<CatalogViewMode>(
   203→    initialFilters.view === "list" ? "list" : "grid",
   204→  );
   205→  const [page, setPage] = useState(Math.max(1, initialFilters.page ?? 1));
   206→  const [filtersOpen, setFiltersOpen] = useState(false);
   207→
   208→  /* تغییر فیلتر همیشه به صفحهٔ اول برمی‌گردد */
   209→  const updateFilters = (patch: Partial<CatalogFilterState>) => {
   210→    setFilters((prev) => ({ ...prev, ...patch }));
   211→    setPage(1);
   212→  };
   213→
   214→  const resetFilters = () => {
   215→    setFilters({
   216→      category: null,
   217→      brands: [],
   218→      min: null,
   219→      max: null,
   220→      inStock: false,
   221→      discount: false,
   222→      q: null,
   223→    });
   224→    setPage(1);
   225→  };
   226→
   227→  /* ── فیلتر + سورت (روی دادهٔ props) ── */
   228→
   229→  const brandTitleBySlug = useMemo(
   230→    () => new Map(brands.map((brand) => [brand.slug, brand.title])),
   231→    [brands],
   232→  );
   233→
   234→  const filtered = useMemo(() => {
   235→    const query = filters.q?.trim();
   236→    const needle = query ? normalizeText(query) : null;
   237→    const categorySlugs = filters.category
   238→      ? collectCategorySlugs(categories, filters.category)
   239→      : null;
   240→    const selectedBrandTitles = new Set(
   241→      filters.brands
   242→        .map((slug) => brandTitleBySlug.get(slug))
   243→        .filter((title): title is string => title != null),
   244→    );
   245→
   246→    const result = products.filter((product) => {
   247→      if (categorySlugs) {
   248→        const productCategory = categoryByProduct[product.slug] ?? null;
   249→        if (productCategory == null || !categorySlugs.has(productCategory)) {
   250→          return false;
   251→        }
   252→      }
   253→      if (
   254→        selectedBrandTitles.size > 0 &&
   255→        !(product.brandTitle != null && selectedBrandTitles.has(product.brandTitle))
   256→      ) {
   257→        return false;
   258→      }
   259→      if (filters.min != null && product.price < filters.min) return false;
   260→      if (filters.max != null && product.price > filters.max) return false;
   261→      if (filters.inStock && !product.inStock) return false;
   262→      if (filters.discount && product.comparePrice == null) return false;
   263→      if (needle != null && needle !== "" && !normalizeText(product.title).includes(needle)) {
   264→        return false;
   265→      }
   266→      return true;
   267→    });
   268→
   269→    switch (sort) {
   270→      case "cheapest":
   271→        return [...result].sort((a, b) => a.price - b.price);
   272→      case "expensive":
   273→        return [...result].sort((a, b) => b.price - a.price);
   274→      /* آرایهٔ preview قدیمی→جدید است؛ newest یعنی وارونهٔ آن */
   275→      case "newest":
   276→        return [...result].reverse();
   277→      case "popular":
   278→        return result;
   279→    }
   280→  }, [products, categories, categoryByProduct, brandTitleBySlug, filters, sort]);
   281→
   282→  /* ── صفحه‌بندی ── */
   283→
   284→  const totalCount = filtered.length;
   285→  const totalPages = Math.max(1, Math.ceil(totalCount / PAGE_SIZE));
   286→  const safePage = Math.min(page, totalPages);
   287→  const pageItems = useMemo(
   288→    () => filtered.slice((safePage - 1) * PAGE_SIZE, safePage * PAGE_SIZE),
   289→    [filtered, safePage],
   290→  );
   291→
   292→  const goToPage = (next: number) => {
   293→    setPage(Math.min(Math.max(1, next), totalPages));
   294→    window.scrollTo({ top: 0 });
   295→  };
   296→
   297→  /* ── همگام‌سازی با URL (فقط مقادیر غیرپیش‌فرض) ── */
   298→
   299→  useEffect(() => {
   300→    const params = new URLSearchParams();
   301→    if (filters.category) params.set("category", filters.category);
   302→    for (const slug of filters.brands) params.append("brand", slug);
   303→    const query = filters.q?.trim();
   304→    if (query) params.set("q", toEnDigits(query));
   305→    if (sort !== "newest") params.set("sort", sort);
   306→    if (view !== "grid") params.set("view", view);
   307→    if (safePage > 1) params.set("page", String(safePage));
   308→    if (filters.min != null) params.set("min", String(filters.min));
   309→    if (filters.max != null) params.set("max", String(filters.max));
   310→    if (filters.inStock) params.set("inStock", "1");
   311→    if (filters.discount) params.set("discount", "1");
   312→    const queryString = params.toString();
   313→    router.replace(queryString ? `${pathname}?${queryString}` : pathname, {
   314→      scroll: false,
   315→    });
   316→  }, [filters, sort, view, safePage, pathname, router]);
   317→
   318→  /* ── مشتقات نمایشی ── */
   319→
   320→  const activeCategoryTitle = filters.category
   321→    ? findCategoryTitle(categories, filters.category)
   322→    : null;
   323→
   324→  const activeChips = useMemo<ActiveChip[]>(() => {
   325→    const chips: ActiveChip[] = [];
   326→    if (filters.category != null) {
   327→      chips.push({
   328→        key: "category",
   329→        label: activeCategoryTitle ?? filters.category,
   330→        clear: () => updateFilters({ category: null }),
   331→      });
   332→    }
   333→    for (const slug of filters.brands) {
   334→      chips.push({
   335→        key: `brand-${slug}`,
   336→        label: brandTitleBySlug.get(slug) ?? slug,
   337→        clear: () =>
   338→          updateFilters({
   339→            brands: filters.brands.filter((item) => item !== slug),
   340→          }),
   341→      });
   342→    }
   343→    if (filters.min != null) {
   344→      chips.push({
   345→        key: "min",
   346→        label: `از ${formatTomanPlain(filters.min)}`,
   347→        clear: () => updateFilters({ min: null }),
   348→      });
   349→    }
   350→    if (filters.max != null) {
   351→      chips.push({
   352→        key: "max",
   353→        label: `تا ${formatTomanPlain(filters.max)}`,
   354→        clear: () => updateFilters({ max: null }),
   355→      });
   356→    }
   357→    if (filters.inStock) {
   358→      chips.push({
   359→        key: "inStock",
   360→        label: "فقط موجود",
   361→        clear: () => updateFilters({ inStock: false }),
   362→      });
   363→    }
   364→    if (filters.discount) {
   365→      chips.push({
   366→        key: "discount",
   367→        label: "فقط تخفیف‌دار",
   368→        clear: () => updateFilters({ discount: false }),
   369→      });
   370→    }
   371→    const query = filters.q?.trim();
   372→    if (query) {
   373→      chips.push({
   374→        key: "q",
   375→        label: `جستجو: «${query}»`,
   376→        clear: () => updateFilters({ q: null }),
   377→      });
   378→    }
   379→    return chips;
   380→  }, [filters, activeCategoryTitle, brandTitleBySlug]);
   381→
   382→  const hasActiveFilters = activeChips.length > 0;
   383→
   384→  const filterPanel = (
   385→    <FilterPanel
   386→      categories={categories}
   387→      brands={brands}
   388→      filters={filters}
   389→      onChange={updateFilters}
   390→      onReset={resetFilters}
   391→    />
   392→  );
   393→
   394→  return (
   395→    <div>
   396→      {/* ── تولبار — شناورِ شیشه‌ای در دسکتاپ ── */}
   397→      <div
   398→        className="rounded-2xl border border-border/70 bg-card px-4 py-3 shadow-card lg:sticky lg:z-30 lg:bg-background/85 lg:backdrop-blur-md"
   399→        style={{ top: `${DESKTOP_HEADER_H - 1}px` }}
   400→      >
   401→        <div className="flex flex-wrap items-center gap-3">
   402→          {/* فیلتر موبایل — دراور پایین‌کش */}
   403→          <Drawer open={filtersOpen} onOpenChange={setFiltersOpen}>
   404→            <DrawerTrigger asChild>
   405→              <Button
   406→                type="button"
   407→                variant="outline"
   408→                className="h-11 gap-2 rounded-full border-border/70 lg:hidden"
   409→              >
   410→                <SlidersHorizontal className="size-4" aria-hidden="true" />
   411→                فیلترها
   412→                {hasActiveFilters ? (
   413→                  <span className="flex size-5 items-center justify-center rounded-full bg-gold text-[10px] font-black text-gold-foreground">
   414→                    {toFa(activeChips.length)}
   415→                  </span>
   416→                ) : null}
   417→              </Button>
   418→            </DrawerTrigger>
   419→            <DrawerContent className="rounded-t-3xl!">
   420→              <DrawerHeader className="text-start">
   421→                <DrawerTitle className="text-base font-bold">
   422→                  فیلترها
   423→                </DrawerTitle>
   424→                <DrawerDescription className="sr-only">
   425→                  فیلتر محصولات بر اساس دسته‌بندی، برند، قیمت و موجودی
   426→                </DrawerDescription>
   427→              </DrawerHeader>
   428→              <div className="flex-1 overflow-y-auto px-4 pb-2">
   429→                {filterPanel}
   430→              </div>
   431→              <DrawerFooter className="border-t border-border/60">
   432→                {hasActiveFilters ? (
   433→                  <Button
   434→                    type="button"
   435→                    variant="ghost"
   436→                    className="h-9 text-destructive hover:text-destructive"
   437→                    onClick={resetFilters}
   438→                  >
   439→                    حذف همهٔ فیلترها
   440→                  </Button>
   441→                ) : null}
   442→                <Button
   443→                  type="button"
   444→                  className="h-12 rounded-xl text-base font-bold"
   445→                  onClick={() => setFiltersOpen(false)}
   446→                >
   447→                  مشاهدهٔ {toFa(totalCount)} کالا
   448→                </Button>
   449→              </DrawerFooter>
   450→            </DrawerContent>
   451→          </Drawer>
   452→
   453→          {/* شمار نتیجه */}
   454→          <p className="hidden text-sm text-muted-foreground lg:block">
   455→            {toFa(totalCount)} کالا
   456→          </p>
   457→
   458→          <div className="ms-auto flex items-center gap-2">
   459→            {/* سورت */}
   460→            <Select
   461→              value={sort}
   462→              onValueChange={(value) => {
   463→                const option = SORT_OPTIONS.find((item) => item.key === value);
   464→                if (option) {
   465→                  setSort(option.key);
   466→                  setPage(1);
   467→                }
   468→              }}
   469→            >
   470→              <SelectTrigger
   471→                className="h-11 w-[10.5rem] rounded-full border-border/70"
   472→                aria-label="مرتب‌سازی محصولات"
   473→              >
   474→                <SelectValue placeholder="مرتب‌سازی" />
   475→              </SelectTrigger>
   476→              <SelectContent>
   477→                {SORT_OPTIONS.map((option) => (
   478→                  <SelectItem key={option.key} value={option.key}>
   479→                    {option.label}
   480→                  </SelectItem>
   481→                ))}
   482→              </SelectContent>
   483→            </Select>
   484→
   485→            {/* نمای شبکه/فهرست */}
   486→            <div
   487→              className="flex items-center gap-0.5 rounded-full border border-border/70 bg-card p-1"
   488→              role="group"
   489→              aria-label="نوع نمایش محصولات"
   490→            >
   491→              <Button
   492→                type="button"
   493→                variant="ghost"
   494→                size="icon"
   495→                className={cn(
   496→                  "size-9 rounded-full",
   497→                  view === "grid"
   498→                    ? "bg-accent text-primary"
   499→                    : "text-muted-foreground",
   500→                )}
   501→                onClick={() => setView("grid")}
   502→                aria-label="نمای شبکه‌ای"
   503→                aria-pressed={view === "grid"}
   504→              >
   505→                <LayoutGrid className="size-4" aria-hidden="true" />
   506→              </Button>
   507→              <Button
   508→                type="button"
   509→                variant="ghost"
   510→                size="icon"
   511→                className={cn(
   512→                  "size-9 rounded-full",
   513→                  view === "list"
   514→                    ? "bg-accent text-primary"
   515→                    : "text-muted-foreground",
   516→                )}
   517→                onClick={() => setView("list")}
   518→                aria-label="نمای فهرستی"
   519→                aria-pressed={view === "list"}
   520→              >
   521→                <List className="size-4" aria-hidden="true" />
   522→              </Button>
   523→            </div>
   524→          </div>
   525→        </div>
   526→      </div>
   527→
   528→      {/* ── چیپ‌های فیلتر فعال ── */}
   529→      {hasActiveFilters ? (
   530→        <div
   531→          className="mt-4 flex flex-wrap items-center gap-2"
   532→          aria-label="فیلترهای فعال"
   533→        >
   534→          {activeChips.map((chip) => (
   535→            <button
   536→              key={chip.key}
   537→              type="button"
   538→              onClick={chip.clear}
   539→              className="group inline-flex h-8 items-center gap-1.5 rounded-full border border-gold/40 bg-gold-soft/25 px-3 text-xs font-medium text-foreground transition-colors hover:bg-gold-soft/50 dark:bg-gold-soft/15 dark:hover:bg-gold-soft/25"
   540→            >
   541→              {chip.label}
   542→              <X
   543→                className="size-3.5 text-muted-foreground transition-colors group-hover:text-destructive"
   544→                aria-hidden="true"
   545→              />
   546→            </button>
   547→          ))}
   548→          <button
   549→            type="button"
   550→            onClick={resetFilters}
   551→            className="mr-1 text-xs font-medium text-destructive transition-colors hover:text-destructive/80"
   552→          >
   553→            حذف همه
   554→          </button>
   555→        </div>
   556→      ) : null}
   557→
   558→      {/* ── بدنه: سایدبار فیلتر + محتوا ── */}
   559→      <div className="mt-6 flex gap-8">
   560→        <aside
   561→          className="hidden w-64 shrink-0 lg:block"
   562→          aria-label="فیلترهای محصولات"
   563→        >
   564→          <div

   586→                </p>

   594→                  className="h-10 rounded-full px-6"
   595→                  onClick={resetFilters}
   596→                >
   597→                  حذف فیلترها
   598→                </Button>
   599→              ) : null}
   600→            </div>
   601→          ) : view === "grid" ? (
   602→            <ProductGrid products={pageItems} />
   603→          ) : (
   604→            <div className="space-y-3">
   605→              {pageItems.map((product) => (
   606→                <CatalogListCard key={product.id} product={product} />
   607→              ))}
   608→            </div>

   610→
   611→          {totalPages > 1 ? (
   612→            <Pagination className="mt-10">

   615→                <PaginationItem>

   622→                      safePage === 1 && "pointer-events-none opacity-40",
   623→                    )}

   626→                      goToPage(safePage - 1);

   643→                        className="size-11"
   644→                        onClick={(event) => {
   645→                          event.preventDefault();
   646→                          goToPage(item);
   647→                        }}
   648→                      >
   649→                        {toFa(item)}
   650→                      </PaginationLink>
   651→                    </PaginationItem>
   652→                  ),
   653→                )}
   654→
   655→                <PaginationItem>
   656→                  <PaginationLink
   657→                    href="#"
   658→                    aria-label="صفحهٔ بعدی"
   659→                    aria-disabled={safePage === totalPages}
   660→                    className={cn(
   661→                      "size-11",
   662→                      safePage === totalPages && "pointer-events-none opacity-40",
   663→                    )}
   664→                    onClick={(event) => {
   665→                      event.preventDefault();
   666→                      goToPage(safePage + 1);
   667→                    }}
   668→                  >
   669→                    <ChevronLeft className="size-4" aria-hidden="true" />
   670→                  </PaginationLink>
   671→                </PaginationItem>
   672→              </PaginationContent>
   673→            </Pagination>
   674→          ) : null}
   675→        </div>
   676→      </div>
   677→    </div>
   678→  );
   679→}
